      * It tracks user intent during a session - don't have to keep repeating the intent of what the session is all about.
      * supporting multi-step reasoning and workflows - without having to keep repeating some of the reasoning that happened before in some of the answers that you got from an agent or from an LLM before

        public void ChatHistory()
        {
            var aiClient = new AzureOpenAIClient(new Uri("AzureOpenAI:Endpoint"), new ApiKeyCredential("AzureOpenAI:APIKey"));
           
            //var cosmosConnectionString = config["CosmosDb:ConnectionString"];

            var database = cosmosClient.GetDatabase("CosmosDBConference2026");

            var historyProvider = new CosmosChatHistoryProvider(
                cosmosClient: cosmosClient,
                databaseId: database.Id,
                containerId: container.Id,
                stateInitializer: _=> new CosmosChatHistoryProvider.State(
                    conversationId: conversationId,
                    tenantId: "default-tenant",
                    userId: "default-user"
                 ),
                ownsClient: false
            );

            var agentOptions = new ChatClientAgentOptions();
            agentOptions.ChatHistoryProvider = historyProvider;

            );

            #pragma warning restore MAAI001
