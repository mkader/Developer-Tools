   CosmosChatHistory.cs
   using Azure.AI.OpenAI;
   using Microsoft.Agents;
   using Microsoft.Agents.AI;
   using Microsoft.Agents.AI.Compaction;
   using Microsoft.Azure.Cosmos;
   using Microsoft.Extensions.AI;
   using Microsoft.Extensions.Configuration;
   using OpenAI;
   using OpenAI.Chat;
   using System.ClientModel;
   
   
   public class CosmosChatHistory
   {
   
       private readonly AzureOpenAIClient _aiClient;
       private readonly CosmosClient _cosmosClient;
       private readonly string _databaseId;
       private readonly string _containerId;
       private readonly string _deploymentName;
   
       public CosmosChatHistory(IConfiguration config)
       {
           string endpoint = config["AzureOpenAI:Endpoint"];
           string apiKey = config["AzureOpenAI:APIKey"];
           _deploymentName = config["AzureOpenAI:DeploymentName"];
           string conString = config["CosmosDbChatHistory:ConnectionString"];
           _databaseId = config["CosmosDbChatHistory:DatabaseId"];
   
           _aiClient = new AzureOpenAIClient(new Uri(endpoint), new ApiKeyCredential(apiKey));
   
           _cosmosClient = new CosmosClient(conString);
   
           _containerId = "chathistory1";
       }
   
       public AIAgent CreateAgent(string conversationId, string tenantId = "default-tenant", string userId = "default-user")
       {
           var database = _cosmosClient.GetDatabase(_databaseId);
           var container = database.GetContainer(_containerId);
   
           var historyProvider = new CosmosChatHistoryProvider(
               cosmosClient: _cosmosClient,
               databaseId: database.Id,
               containerId: container.Id,
               stateInitializer: _ => new CosmosChatHistoryProvider.State(
                   conversationId: conversationId,
                   tenantId: tenantId,
                   userId: userId
               ),
               ownsClient: false
           );
   
           #pragma warning disable MAAI001
           var compactionStrategy = new SummarizationCompactionStrategy(
               chatClient: _aiClient.GetChatClient(_deploymentName).AsIChatClient(),
               trigger: CompactionTriggers.MessagesExceed(8)
           );
   
           var compactionProvider = new CompactionProvider(compactionStrategy);
           #pragma warning restore MAAI001
   
           var agentOptions = new ChatClientAgentOptions();
           agentOptions.ChatHistoryProvider = historyProvider;
           agentOptions.AIContextProviders = [compactionProvider];
   
           AIAgent agent = _aiClient.GetChatClient(_deploymentName).AsAIAgent(agentOptions);
   
           return agent;
       }
   
       public async Task<string> SendMessageAsync(AIAgent agent, AgentSession session, string message)
       {
           AgentResponse response = await agent.RunAsync(message, session);
           return response.ToString();
       }
   
       public async Task Run(IConfiguration config)
       {
           string endpoint = config["AzureOpenAI:Endpoint"];
           string apiKey = config["AzureOpenAI:APIKey"];
           string deploymentName = config["AzureOpenAI:DeploymentName"];
           string conString = config["CosmosDbChatHistory:ConnectionString"];
           string databaseId = config["CosmosDbChatHistory:DatabaseId"];
   
           var cosmosClient = new CosmosClient(conString);
   
           var database = cosmosClient.GetDatabase(databaseId);
           var container = database.GetContainer("chathistory1");
   
           string conversationId = "LinoConversation";
   
           var historyProvider = new CosmosChatHistoryProvider(
               cosmosClient: cosmosClient,
               databaseId: database.Id,
               containerId: container.Id,
               stateInitializer: _ => new CosmosChatHistoryProvider.State(
                   conversationId: conversationId,
                   tenantId: "default-tenant",
                   userId: "default-user"
               ),
               ownsClient: false
           );
   
           var aiClient = new AzureOpenAIClient(new Uri(endpoint), new ApiKeyCredential(apiKey));
   
   #pragma warning disable MAAI001
           var compactionStrategy = new SummarizationCompactionStrategy(
               chatClient: aiClient.GetChatClient(deploymentName).AsIChatClient(),
               trigger: CompactionTriggers.MessagesExceed(8)
           );
   
           var compactionProvider = new CompactionProvider(compactionStrategy);
   #pragma warning restore MAAI001
   
           var agentOptions = new ChatClientAgentOptions();
           agentOptions.ChatHistoryProvider = historyProvider;
           agentOptions.AIContextProviders = [compactionProvider];
   
           string question = "what are the top 10 Movies according to IMDB?";
           question = "ok can you give the top 5 only";
   
           AIAgent agent = aiClient.GetChatClient(deploymentName).AsAIAgent(agentOptions);
   
           AgentSession session = await agent.CreateSessionAsync();
   
           AgentResponse response = await agent.RunAsync(question, session);
   
           Console.WriteLine(response);
       }
   
       public void Dispose()
       {
           _cosmosClient?.Dispose();
       }
   }

## why is semantic caching important?
* LLMs are definitely the most expensive
  * because the tokens that have to be paid back and forth.
  * For dev/testing 2 or 3 people, it would end up being just pennies.
  * But in production, we have done the implementation for a university, 25,000 students asking questions using the chat AI bot
  * it can accumulate definitely millions or billions of tokens per months
  * it will have a cost associated with it. So cashing is very important.
  * 1st person asks how many classes for chemistry and biology in this semester
  * semantic cache - meaning the question doesn't have to even be exact the semantic meaning because I'm sure a 1000 students will ask same questions very differently way.
  * if the meaning of that is already saved in the Cosmos DB for instance, and I have a vector for semantic meaning/caching based on that, then the other 999 students will not have to pay money or use tokens.
  * It will just get really fast in Cosmos DB. That would be less than 10 millisecond for the response.
  * So it would be extremely good performance
* traditional cache requires an exact match only for the questions like we used like command in SQL databases but that's not going to be very useful
* work on so many different similar questions at the same time and that is what semantic caching is all about is about meaning match not exact match as well.

 * <img width="400" height="250" alt="image" src="https://github.com/user-attachments/assets/c9d9bc34-c37c-4486-b8bf-5073402aea1f" />
 * User questions like "What are the what are the classes offered for biology and chemistry"
 * immediately the questions will be embedded. create 1536 different floats (some models have high 3072 or local olama) that explain what this question is all about.
 * I'm going into the vector store into one of the containers (cosmos db containers).
 * to see any similarities in db, the similarity score evaluation - something above 0.3 or 0.4 or 0.7.
 * You get to specify what you want from the db how similar you want them to be.
 * if it finds it, that's a cash hit - no llm/money involved, return in Cosmos DB less than 10 millisecond the stored answers.
 * if it's not, if it could not find something similar above that score similarity evaluation, then we need to call the LLM. wait a little bit longer until the LM respond.

using Azure.AI.OpenAI;
using DocumentFormat.OpenXml.Office.SpreadSheetML.Y2023.MsForms;
using Microsoft.Agents;
using Microsoft.Agents.AI;
using Microsoft.Agents.AI.Compaction;
using Microsoft.Azure.Cosmos;
using Microsoft.Extensions.AI;
using Microsoft.Extensions.Configuration;
using OpenAI;
using OpenAI.Chat;
using System.ClientModel;
using System.Text.Json.Serialization;

public class CosmosChatHistory
{
     public async Task SemanticCache(IConfiguration config){

        string endpoint = config["AzureOpenAI:Endpoint"];
        string apiKey = config["AzureOpenAI:APIKey"];
        string deploymentName = config["AzureOpenAI:DeploymentName"];
        string conString = config["CosmosDbChatHistory:ConnectionString"];
        string databaseId = config["CosmosDbChatHistory:DatabaseId"];

        var cosmosClient = new CosmosClient(conString);

        var database = cosmosClient.GetDatabase(databaseId);
        var container = database.GetContainer("cache");

        var aiClient = new AzureOpenAIClient(new Uri(endpoint), new ApiKeyCredential(apiKey));
        
        IEmbeddingGenerator<string, Embedding<float>> embeddingGenerator = 
            _aiClient.GetEmbeddingClient("text-embedding-3-small").AsIEmbeddingGenerator();
        
        string question = "what are the top 10 Movies according to IMDB?";

        var embedding = await embeddingGenerator.GenerateAsync(question);

        AIAgent agent = aiClient.GetChatClient(deploymentName).AsAIAgent();

        AgentSession session = await agent.CreateSessionAsync();

        var cached = await FindSimilarAsync(container, embedding.Vector.ToArray());

        if(cached != null)
        {
            Console.WriteLine("CACHE HIT: Found similar question in cache");
            Console.WriteLine($"Question: {cached.question}");
            Console.WriteLine($"Answer: {cached.answer}"); 
            Console.WriteLine();
            return;
        }

        Console.WriteLine("CACHE MISS: Calling LLM");

        AgentResponse response = await agent.RunAsync(question, session);

        var item = new SemanticCacheItem
        {
            id = Guid.NewGuid().ToString(),
            pk = "cache",
            question = question,
            answer = response.Text,
            embedding = embedding.Vector.ToArray()
        };

        await container.CreateItemAsync(item);

        Console.WriteLine(response);
    }

    public async Task<SemanticCacheItem?> FindSimilarAsync(Container container, float[] embedding)
    {
        var query = new QueryDefinition(@"
            SELECT TOP 1 c.id, c.question, c.answer, VectorDistance(c.embedding, @embedding) AS similarity
            FROM c
            WHERE c.pk = @pk AND VectorDistance(c.embedding, @embedding) > 0.3
            ORDER BY VectorDistance(c.embedding, @embedding)
        ")
        .WithParameter("@embedding", embedding)
        .WithParameter("@pk", "cache");

        using var iterator = container.GetItemQueryIterator<SemanticCacheItem>(query);

        while (iterator.HasMoreResults)
        {
            foreach (var item in await iterator.ReadNextAsync())
            {
                return item;
            }
        }

        return null;
    }

    public class SemanticCacheItem
    {
        [JsonPropertyName("id")]
        public string id { get; set; } = string.Empty;

        [JsonPropertyName("pk")]
        public string pk { get; set; } = "cache";

        [JsonPropertyName("question")]
        public string question { get; set; } = string.Empty;

        [JsonPropertyName("answer")]
        public string answer { get; set; } = string.Empty;

        [JsonPropertyName("embedding")]
        public float[] embedding { get; set; } = [];

        //[JsonPropertyName("similarity")]
        //public double? Similarity { get; set; }
    }

}
